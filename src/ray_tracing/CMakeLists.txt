set(RAY_TRACING_SUBPROJECT_NAME "ray_tracing" CACHE INTERNAL "")

add_executable(${RAY_TRACING_SUBPROJECT_NAME}
    "main.cpp"
)

target_link_libraries(${RAY_TRACING_SUBPROJECT_NAME} PUBLIC ${CORE_LIB_NAME})

set_target_properties(${RAY_TRACING_SUBPROJECT_NAME} PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${CMAKE_BUILD_TYPE}/${RAY_TRACING_SUBPROJECT_NAME}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${CMAKE_BUILD_TYPE}/${RAY_TRACING_SUBPROJECT_NAME}"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${CMAKE_BUILD_TYPE}/${RAY_TRACING_SUBPROJECT_NAME}"
)

foreach(OUTPUTCONFIG IN LISTS CMAKE_CONFIGURATION_TYPES)
    string(TOUPPER ${OUTPUTCONFIG} OUTPUTCONFIG_UPPER)
    set_target_properties(${RAY_TRACING_SUBPROJECT_NAME} PROPERTIES
        ARCHIVE_OUTPUT_DIRECTORY_${OUTPUTCONFIG_UPPER} "${CMAKE_BINARY_DIR}/${OUTPUTCONFIG}/${RAY_TRACING_SUBPROJECT_NAME}"
        LIBRARY_OUTPUT_DIRECTORY_${OUTPUTCONFIG_UPPER} "${CMAKE_BINARY_DIR}/${OUTPUTCONFIG}/${RAY_TRACING_SUBPROJECT_NAME}"
        RUNTIME_OUTPUT_DIRECTORY_${OUTPUTCONFIG_UPPER} "${CMAKE_BINARY_DIR}/${OUTPUTCONFIG}/${RAY_TRACING_SUBPROJECT_NAME}"
    )
endforeach()
